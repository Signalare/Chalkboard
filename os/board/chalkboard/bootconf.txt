# Production EEPROM bootloader configuration (OS-011, BOOT-001).
# Applies to BCM2711 and BCM2712 boards with on-board EEPROM.

[all]
# Disable bootloader UART logging in production to save boot time.
BOOT_UART=0

# Boot order: microSD first, then USB mass storage, then restart (0xf41).
BOOT_ORDER=0xf41

# Save ~1.0 s by skipping USB power-off cycle when booting from SD.
USB_MSD_PWR_OFF_TIME=0

# No HDMI diagnostics screen and no network install prompt (BOOT-007).
DISABLE_HDMI=1
NET_INSTALL_AT_POWER_ON=0

WAKE_ON_GPIO=1
POWER_OFF_ON_HALT=0
